import { useState } from "react";
import { useDispatch, useSelector } from "react-redux";
import { Link, useNavigate } from "react-router-dom";
import { clearCart } from "../../store/slices/cartSlice";
import { createOrder } from "../../services/orderService";
import { getAddresses } from "../../services/addressService";
import { getAvailableStock } from "../../services/inventoryService";

const emptyAddress = {
  fullName: "",
  phone: "",
  house: "",
  street: "",
  city: "Hyderabad",
  pincode: "",
};

function Checkout() {
  const dispatch = useDispatch();
  const navigate = useNavigate();

  const cartItems = useSelector((state) => state.cart.items);
  const user = useSelector((state) => state.auth.user);

  const [savedAddresses] = useState(() =>
    user?.id ? getAddresses(user.id) : []
  );

  const defaultAddress =
    savedAddresses.find((address) => address.isDefault) ||
    savedAddresses[0];

  const [selectedAddressId, setSelectedAddressId] = useState(
    defaultAddress?.id || "new"
  );

  const [address, setAddress] = useState(
    defaultAddress
      ? {
          fullName: defaultAddress.fullName,
          phone: defaultAddress.phone,
          house: defaultAddress.house,
          street: defaultAddress.street,
          city: defaultAddress.city,
          pincode: defaultAddress.pincode,
        }
      : {
          ...emptyAddress,
          fullName: user?.name || "",
          phone: user?.phone || "",
        }
  );

  const [instructions, setInstructions] = useState("");
  const [error, setError] = useState("");
  const [isSubmitting, setIsSubmitting] = useState(false);

  const subtotal = cartItems.reduce(
    (total, item) => total + item.price * item.quantity,
    0
  );

  const totalItems = cartItems.reduce(
    (total, item) => total + item.quantity,
    0
  );

  const formatPrice = (amount) =>
    `â‚¹${amount.toLocaleString("en-IN")}`;

  const handleAddressSelection = (event) => {
    const selectedId = event.target.value;
    setSelectedAddressId(selectedId);

    if (selectedId === "new") {
      setAddress({
        ...emptyAddress,
        fullName: user?.name || "",
        phone: user?.phone || "",
      });
      return;
    }

    const selected = savedAddresses.find(
      (item) => item.id === selectedId
    );

    if (selected) {
      setAddress({
        fullName: selected.fullName,
        phone: selected.phone,
        house: selected.house,
        street: selected.street,
        city: selected.city,
        pincode: selected.pincode,
      });
    }
  };

  const handleChange = (event) => {
    const { name, value } = event.target;

    setAddress((previous) => ({
      ...previous,
      [name]: value,
    }));

    setSelectedAddressId("new");
  };

  const handlePlaceOrder = (event) => {
    event.preventDefault();

    if (isSubmitting) return;

    setError("");

    if (!user?.id) {
      setError("Please log in before placing an order.");
      return;
    }

    if (cartItems.length === 0) {
      setError("Your cart is empty.");
      return;
    }

    if (!/^[6-9]\d{9}$/.test(address.phone)) {
      setError("Enter a valid 10-digit Indian mobile number.");
      return;
    }

    if (!/^\d{6}$/.test(address.pincode)) {
      setError("Enter a valid 6-digit PIN code.");
      return;
    }

    // Final stock check before submitting the order.
    const unavailableItems = cartItems.filter(
      (item) => item.quantity > getAvailableStock(item.id)
    );

    if (unavailableItems.length > 0) {
      setError(
        `Stock changed for: ${unavailableItems
          .map((item) => item.title)
          .join(", ")}. Please update your cart.`
      );
      return;
    }

    setIsSubmitting(true);

    try {
      const order = createOrder({
        userId: user.id,
        items: cartItems,
        address,
        instructions,
        paymentMethod: "PAY_ON_DELIVERY",
      });

      dispatch(clearCart());

      navigate(`/orders/${order.id}`, {
        state: { orderPlaced: true },
      });
    } catch (error) {
      setError(error.message || "Unable to place your order.");
      setIsSubmitting(false);
    }
  };

  if (cartItems.length === 0) {
    return (
      <main className="checkout-page">
        <section className="checkout-empty">
          <h2>Your cart is empty</h2>
          <p>Add some products before proceeding to checkout.</p>
          <Link to="/products" className="checkout-primary-button">
            Explore Products
          </Link>
        </section>
      </main>
    );
  }

  return (
    <main className="checkout-page">
      <div className="checkout-heading">
        <h1>Checkout</h1>
        <p>Almost there! Confirm your delivery details.</p>
      </div>

      <form onSubmit={handlePlaceOrder}>
        <div className="checkout-layout">
          <div className="checkout-main">
            <section className="checkout-card">
              <h2>1. Delivery Address</h2>

              <div className="checkout-saved-address">
                {savedAddresses.length > 0 ? (
                  <>
                    <label htmlFor="saved-address">
                      Choose a saved address
                    </label>

                    <select
                      id="saved-address"
                      value={selectedAddressId}
                      onChange={handleAddressSelection}
                    >
                      {savedAddresses.map((item) => (
                        <option key={item.id} value={item.id}>
                          {item.fullName} â€” {item.city},{" "}
                          {item.pincode}
                          {item.isDefault ? " (Default)" : ""}
                        </option>
                      ))}

                      <option value="new">
                        + Enter a different address
                      </option>
                    </select>
                  </>
                ) : (
                  <p>You don't have any saved addresses yet.</p>
                )}

                <Link to="/addresses" className="manage-address-link">
                  + Add or manage saved addresses â†’
                </Link>
              </div>

              <div className="checkout-form-grid">
                <input
                  name="fullName"
                  value={address.fullName}
                  onChange={handleChange}
                  placeholder="Full name"
                  required
                />

                <input
                  name="phone"
                  type="tel"
                  value={address.phone}
                  onChange={handleChange}
                  placeholder="10-digit mobile number"
                  pattern="[6-9][0-9]{9}"
                  maxLength={10}
                  required
                />

                <input
                  name="house"
                  value={address.house}
                  onChange={handleChange}
                  placeholder="House / Flat / Building"
                  required
                />

                <input
                  name="street"
                  value={address.street}
                  onChange={handleChange}
                  placeholder="Street / Area / Landmark"
                  required
                />

                <input
                  name="city"
                  value={address.city}
                  onChange={handleChange}
                  placeholder="City"
                  required
                />

                <input
                  name="pincode"
                  value={address.pincode}
                  onChange={handleChange}
                  placeholder="6-digit PIN code"
                  pattern="[0-9]{6}"
                  maxLength={6}
                  required
                />
              </div>

              {selectedAddressId !== "new" && (
                <p className="checkout-address-note">
                  Selected saved address. Editing any field will use it
                  as a one-time delivery address for this order.
                </p>
              )}
            </section>

            <section className="checkout-card">
              <h2>2. Delivery Instructions</h2>

              <textarea
                value={instructions}
                onChange={(event) =>
                  setInstructions(event.target.value)
                }
                placeholder="Any instructions for the delivery partner? (Optional)"
                rows={3}
              />
            </section>

                Pay on delivery (demo)

            {cartItems.map((item) => {

                    <strong>{item.title}</strong>
                    <p>
                      {item.quantity} Ã— {formatPrice(item.price)}
                    </p>

                    <p
                      style={{
                        color: hasEnoughStock ? "green" : "red",

import {
  getInventory,
  adjustStockBatch,
} from "./inventoryService";

const ORDERS_STORAGE_KEY = "quickkart-orders";

const ORDER_STAGES = [
  "PLACED",
  "CONFIRMED",
  "SHIPPED",
  "OUT_FOR_DELIVERY",
  "DELIVERED",
];

const getStoredOrders = () => {
  try {
    return JSON.parse(
      localStorage.getItem(ORDERS_STORAGE_KEY) || "[]"
    );
  } catch (error) {
    console.error("Unable to read orders:", error);
    return [];
  }
};

const saveOrders = (orders) => {
  localStorage.setItem(
    ORDERS_STORAGE_KEY,
    JSON.stringify(orders)
  );
};

const addStatusHistory = (order, status) => {
  if (!Array.isArray(order.statusHistory)) {
    order.statusHistory = [];
  }

  order.status = status;

  order.statusHistory.push({
    status,
    timestamp: new Date().toISOString(),
  });
};

const restoreStock = (items) => {
  const changes = items.map((item) => ({
    productId: item.id,
    quantityChange: item.quantity,
  }));

  return adjustStockBatch(changes);
};

// Shared status transition validation
const applyStatusTransition = (order, newStatus) => {
  if (order.status === "CANCELLED") {
    throw new Error("Cancelled orders cannot be updated.");
  }

  if (order.status === "DELIVERED") {
    throw new Error("Delivered orders cannot be updated.");
  }

  const currentIndex = ORDER_STAGES.indexOf(order.status);

  if (currentIndex === -1) {
    throw new Error("Current order status is invalid.");
  }

  const expectedNextStatus = ORDER_STAGES[currentIndex + 1];

  if (newStatus !== expectedNextStatus) {
    throw new Error(
      `Invalid status transition. Expected: ${expectedNextStatus || "none"}.`
    );
  }

  addStatusHistory(order, newStatus);
};

// Create order
export const createOrder = ({
  userId,
  items,
  address,
  instructions,
  paymentMethod = "PAY_ON_DELIVERY",
}) => {
  if (!userId) {
    throw new Error("User must be logged in to place an order.");
  }

  if (!items || items.length === 0) {
    throw new Error("Cannot create an order with an empty cart.");
  }

  const inventory = getInventory();
  const requestedQuantities = new Map();

  items.forEach((item) => {
    if (
      !Number.isInteger(item.quantity) ||
      item.quantity <= 0 ||
      !Number.isFinite(item.price) ||
      item.price < 0
    ) {
      throw new Error(
        `Invalid quantity or price for ${item.title}.`
      );
    }

    requestedQuantities.set(
      item.id,
      (requestedQuantities.get(item.id) || 0) + item.quantity
    );
  });

  requestedQuantities.forEach((quantity, productId) => {
    const inventoryItem = inventory.find(
      (item) => item.productId === productId
    );

    if (!inventoryItem) {
      throw new Error(
        "Inventory record not found for a cart product."
      );
    }

    const availableStock =
      inventoryItem.stockQuantity -
      inventoryItem.reservedQuantity;

    if (quantity > availableStock) {
      throw new Error(
        `Only ${Math.max(0, availableStock)} units of product ID ${productId} are available. Please update your cart.`
      );
    }
  });

  const subtotal = items.reduce(
    (total, item) => total + item.price * item.quantity,
    0
  );

  const timestamp = new Date().toISOString();

  const order = {
    id: `QK${Date.now().toString().slice(-8)}`,
    userId,
    items: items.map((item) => ({ ...item })),
    address: { ...address },
    instructions,
    paymentMethod,
    subtotal,
    deliveryCharge: 0,
    total: subtotal,
    status: "PLACED",
    createdAt: timestamp,
    statusHistory: [
      {
        status: "PLACED",
        timestamp,
      },
    ],
  };

  const stockChanges = [...requestedQuantities].map(
    ([productId, quantity]) => ({
      productId,
      quantityChange: -quantity,
    })
  );

  adjustStockBatch(stockChanges);

  try {
    const orders = getStoredOrders();
    orders.unshift(order);
    saveOrders(orders);
  } catch (error) {
    restoreStock(order.items);
    throw new Error(
      "Unable to save your order. Please try again."
    );
  }

  return order;
};

// Customer: get their orders
export const getOrdersByUser = (userId) => {
  return getStoredOrders().filter(
    (order) => order.userId === userId
  );
};

// Admin: get all customer orders
export const getAllOrders = () => {
  return getStoredOrders();
};

// Customer: get one of their orders
export const getOrderById = (orderId, userId) => {
  return (
    getStoredOrders().find(
      (order) =>
        order.id === orderId &&
        order.userId === userId
    ) || null
  );
};

// Customer cancellation
export const cancelOrder = (orderId, userId) => {
  const orders = getStoredOrders();

  const order = orders.find(
    (item) =>
      item.id === orderId &&
      item.userId === userId
  );

  if (!order) {
    throw new Error("Order not found.");
  }

  if (!["PLACED", "CONFIRMED"].includes(order.status)) {
    throw new Error(
      "This order can no longer be cancelled."
    );
  }

  restoreStock(order.items);
  addStatusHistory(order, "CANCELLED");

  try {
    saveOrders(orders);
  } catch (error) {
    adjustStockBatch(
      order.items.map((item) => ({
        productId: item.id,
        quantityChange: -item.quantity,
      }))
    );

    throw new Error(
      "Unable to cancel the order. Please try again."
    );
  }

  return order;
};

// Customer: simulated delivery status update
export const updateOrderStatus = (
  orderId,
  userId,
  newStatus
) => {
  const orders = getStoredOrders();

  const order = orders.find(
    (item) =>
      item.id === orderId &&
      item.userId === userId
  );

  if (!order) {
    throw new Error("Order not found.");
  }

  applyStatusTransition(order, newStatus);
  saveOrders(orders);

  return order;
};

// Admin: update any customer's order
export const updateOrderStatusByAdmin = (
  orderId,
  newStatus
) => {
  const orders = getStoredOrders();

  const order = orders.find(
    (item) => item.id === orderId
  );

  if (!order) {
    throw new Error("Order not found.");
  }

  applyStatusTransition(order, newStatus);
  saveOrders(orders);

  return order;
};

import { products as initialProducts } from "../data/products";

const STORAGE_KEY = "quickkart-products";

export const getProducts = () => {
  try {
    const storedProducts = localStorage.getItem(STORAGE_KEY);

    if (storedProducts !== null) {
      return JSON.parse(storedProducts);
    }

    // First-time initialization
    localStorage.setItem(
      STORAGE_KEY,
      JSON.stringify(initialProducts)
    );

    return initialProducts;
  } catch (error) {
    console.error("Unable to load products:", error);
    return initialProducts;
  }
};

const saveProducts = (products) => {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(products));
};

export const addProduct = (product) => {
  const products = getProducts();

  const newProduct = {
    ...product,
    id: Date.now(),
    rating: Number(product.rating) || 0,
    price: Number(product.price),
    originalPrice: Number(product.originalPrice),
  };

  const updatedProducts = [...products, newProduct];

  saveProducts(updatedProducts);

  return newProduct;
};

export const updateProduct = (productId, updatedData) => {
  const products = getProducts();

  const updatedProducts = products.map((product) =>
    product.id === productId
      ? {
          ...product,
          ...updatedData,
          price: Number(updatedData.price),
          originalPrice: Number(updatedData.originalPrice),
          rating: Number(updatedData.rating) || 0,
        }
      : product
  );

  saveProducts(updatedProducts);

  return updatedProducts;
};

export const deleteProduct = (productId) => {
  const products = getProducts();

  const updatedProducts = products.filter(
    (product) => product.id !== productId
  );

  saveProducts(updatedProducts);

  return updatedProducts;
};
